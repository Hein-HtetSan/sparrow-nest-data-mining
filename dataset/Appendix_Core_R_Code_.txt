2.Mixed-effects model analysis
# Structural Equation Modeling (SEM)
library(nlme)
library(piecewiseSEM)

m01<-lme(fixed = AvaNR2~Altitude+BuildingHeight+nd,
         random = ~ 1 | ID2,data = b5,
         weights =    varIdent(form = ~1 | alt_group_quantile ))
m02<-lme(fixed = DensityR ~ AvaNR2 + BuildingHeight+Altitude,
            random = ~ 1 | ID,data = b5,
            weights = varComb(
              varFixed(~ I(1/(AvaNR2 + 1))),  # fix weight：1/(AvaNR2+1)
              varIdent(form = ~1 | alt_group_quantile) ))
m03<- lme(fixed = NestHeightR~DensityR +AvaNR2 +BuildingHeight +Altitude+nd , 
          random = ~ 1 | Loc,  data = b5,
          weights = varComb(
            varFixed(~ I(DensityR+1 )),  #  fix weight：DensityR
            varIdent(form = ~1 | alt_group_quantile) ))
nest.psem <- psem(m01, m02,m03)
summary(nest.psem)

Annotation: lme4 for hierarchical modeling of altitude-GDP interactions on nest density (manuscript: "mixed-effects models for socioeconomic gradients").
SEM via piecewiseSEM with variable selection (partial R² > 0.3) (manuscript: "SEM for multi-factor pathway analysis").
3. Interaction Effects & Johnson-Neyman Analysis

# Interaction visualization and significance thresholds
library(nlme)
library(interactions)
fit1 <-lme(fixed = DensityR ~ gdpp*Altitude ,
            random = ~ 1 | ID,data = b5,
            weights = varComb(
              varFixed(~ I(1/(AvaNR2 + 1))),  # fix weight：1/(AvaNR2+1)
              varIdent(form = ~1 | alt_group_quantile) ))

johnson_neyman(fit1, pred="gdpp", modx="Altitude")
interact_plot(fit1, pred="gdpp", modx="Altitude", plot.points=T)
Annotation:
Annotation: Calculates altitude-modulated thresholds for GDP effects (manuscript: "GDP suppresses nest density at low altitudes but enhances it at high altitudes") using Johnson-Neyman technique.
